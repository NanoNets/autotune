# Experiment 4 (HF Trainer through the launcher): exact versions, installed on top of the box's
# preinstalled CUDA torch (`python3 -m venv --system-site-packages`). Checked at runtime by
# hf/versions.py; a mismatch fails the run.
#
# torch 2.7.0 (CUDA 12.8 build) is NOT installed from here: it is the box's system torch, and a
# pip torch would replace the CUDA build. transformers 5.15.1 requires torch>=2.5.
#
# transformers / huggingface-hub / tokenizers / safetensors: exactly the repo's uv.lock, and the
# transformers version the source checks in bring_your_own_script.md §2 were read against.
transformers==5.15.1
huggingface-hub==1.28.0
tokenizers==0.22.2
safetensors==0.8.0
# accelerate is not in uv.lock (not a repo dependency). 1.14.0 = the release current when
# transformers 5.15.1 shipped (2026-08-19; 1.15.0 came 2026-09-09). Satisfies transformers'
# accelerate>=1.1.0 and needs torch>=2.0.0.
accelerate==1.14.0
# numpy: three constraints, all found on the box.
#  - >=1.22: transformers 5.15.1 declares numpy>=1.17, but importing Trainer fails below 1.22
#    (data_collator.py:974 subscripts `np.ndarray[...]` at class-definition time). The box's
#    system numpy is 1.21.5.
#  - <2: the box's torch 2.7.0 is the SYSTEM package (/usr/lib/python3/dist-packages), built
#    against the numpy 1.x ABI, as is the system scipy. numpy 2.2.6 was tried: torch's numpy
#    bridge fails ("_ARRAY_API not found") and scipy fails ("numpy.dtype size changed").
#  - <1.25: the box's system scipy (imported by transformers' Trainer) supports numpy
#    >=1.17.3,<1.25.0 and warns on 1.26.4.
#  uv.lock's 2.4.6 / 2.5.2 are out anyway (Python>=3.11; the box has 3.10.12).
#  1.24.4 = the last numpy release below 1.25 (2023-06-26, Python>=3.8). Satisfies all three.
numpy==1.24.4
# pillow: the box's system Pillow 9.0.1 leaks in through --system-site-packages and predates
# `PIL.Image.Resampling` (9.1). transformers' lazy importer swallows the AttributeError and reports
# `ModuleNotFoundError: Could not import module 'PreTrainedModel'`, so Trainer can't import.
# 12.3.0 = the version experiments 4 and 4b ran with (it was installed by hand on that box and
# not pinned here; found when a fresh box failed on 2026-09-25).
pillow==12.3.0
# peft: LoRA experiment (7). Not in uv.lock. 0.20.0 = the release current when transformers
# 5.15.1 shipped (2026-07-28; 0.21.0 came 2026-09-15). Declares torch>=1.13, accelerate>=0.21,
# huggingface-hub>=0.25, python>=3.10: all satisfied. It does not pin transformers, so
# compatibility with 5.15.1 is verified by running, not assumed.
peft==0.20.0

# NOT installed from here, but checked by versions.py because they come from the box's system
# site-packages: torch 2.7.0 (CUDA 12.8) and scipy 1.8.0 (imported by transformers' Trainer).
